cmake_minimum_required(VERSION 3.24)
project(ASMRCLIP VERSION 0.6.15 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "This GUI currently targets Windows 10/11 x64.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(ASMRCLIP WIN32 src/main.cpp src/MainWindow.cpp src/MainWindowPages.cpp src/MainWindowTests.cpp src/ProcessRunner.cpp src/UiControls.cpp src/app.rc)
target_include_directories(ASMRCLIP PRIVATE src third_party)
target_compile_definitions(ASMRCLIP PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_link_libraries(ASMRCLIP PRIVATE comctl32 shell32 ole32 uxtheme gdiplus dwmapi)
if(MSVC)
  target_compile_options(ASMRCLIP PRIVATE /W4 /permissive- /utf-8 /EHsc)
  target_link_options(ASMRCLIP PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_SOURCE_DIR}/src/app.manifest")
  set_property(TARGET ASMRCLIP PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(ASMRCLIP_OUTPUT_DIR "${CMAKE_SOURCE_DIR}/bin" CACHE PATH "GUI executable output directory")
set_target_properties(ASMRCLIP PROPERTIES OUTPUT_NAME "asmrcliper" RUNTIME_OUTPUT_DIRECTORY "${ASMRCLIP_OUTPUT_DIR}")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
  string(TOUPPER ${config} CONFIG_UPPER)
  set_target_properties(ASMRCLIP PROPERTIES OUTPUT_NAME "asmrcliper" RUNTIME_OUTPUT_DIRECTORY_${CONFIG_UPPER} "${ASMRCLIP_OUTPUT_DIR}")
endforeach()
enable_testing()
add_test(NAME gui_program_menu COMMAND ASMRCLIP --test-menu "${CMAKE_CURRENT_BINARY_DIR}/menu-regression.json")
set_tests_properties(gui_program_menu PROPERTIES TIMEOUT 25)
add_test(NAME gui_task_progress COMMAND ASMRCLIP --test-progress "${CMAKE_CURRENT_BINARY_DIR}/progress-regression.json")
set_tests_properties(gui_task_progress PROPERTIES TIMEOUT 25)
add_test(NAME gui_navigation_repaint COMMAND ASMRCLIP --test-navigation "${CMAKE_CURRENT_BINARY_DIR}/navigation-regression.json")
set_tests_properties(gui_navigation_repaint PROPERTIES TIMEOUT 20)
add_test(NAME gui_rendering COMMAND ASMRCLIP --test-rendering "${CMAKE_CURRENT_BINARY_DIR}/rendering-regression.json")
set_tests_properties(gui_rendering PROPERTIES TIMEOUT 20)
add_test(NAME gui_controls COMMAND ASMRCLIP --test-controls "${CMAKE_CURRENT_BINARY_DIR}/controls-regression.json")
set_tests_properties(gui_controls PROPERTIES TIMEOUT 20)
add_test(NAME gui_dropdowns COMMAND ASMRCLIP --test-dropdowns "${CMAKE_CURRENT_BINARY_DIR}/dropdown-regression.json")
set_tests_properties(gui_dropdowns PROPERTIES TIMEOUT 20)
add_test(NAME gui_switch_animation COMMAND ASMRCLIP --test-switches "${CMAKE_CURRENT_BINARY_DIR}/switch-regression.json")
set_tests_properties(gui_switch_animation PROPERTIES TIMEOUT 20)
add_test(NAME gui_settings_autosave COMMAND ASMRCLIP --test-autosave "${CMAKE_CURRENT_BINARY_DIR}/autosave-regression.json")
set_tests_properties(gui_settings_autosave PROPERTIES TIMEOUT 20)
add_executable(process_tests tests/process_tests.cpp src/ProcessRunner.cpp)
target_include_directories(process_tests PRIVATE src)
target_link_libraries(process_tests PRIVATE shell32)
target_compile_definitions(process_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(process_tests PRIVATE /W4 /utf-8)
endif()
add_test(NAME windows_argument_quoting COMMAND process_tests)
if(EXISTS "${CMAKE_SOURCE_DIR}/runtime/venv/Scripts/python.exe")
  set(ASMRCLIP_TEST_PYTHON "${CMAKE_SOURCE_DIR}/runtime/venv/Scripts/python.exe")
elseif(EXISTS "${CMAKE_SOURCE_DIR}/runtime/python/python.exe")
  set(ASMRCLIP_TEST_PYTHON "${CMAKE_SOURCE_DIR}/runtime/python/python.exe")
endif()
if(ASMRCLIP_TEST_PYTHON)
  add_test(NAME process_tree_cancellation COMMAND process_tests "${ASMRCLIP_TEST_PYTHON}")
  add_test(NAME engine_regressions COMMAND "${ASMRCLIP_TEST_PYTHON}" -X utf8 -m unittest discover -s "${CMAKE_SOURCE_DIR}/tests" -p "test_*.py" -v)
endif()
